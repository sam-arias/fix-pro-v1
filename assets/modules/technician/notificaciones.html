// ============================= // NOTIFICACIONES // ============================= let notifications = []; function showNotifications() { const mainContent = document.getElementById('mainContent'); mainContent.innerHTML = `

Notificaciones

${notifications.map(notification => `
${notification.title}
${formatTimeAgo(notification.date)}

${notification.message}

${formatNotificationType(notification.type)} `).join('')}
`; } function viewNotification(notificationId) { const notification = notifications.find(n => n.id === notificationId); if (!notification) return; // Marcar como leída notification.read = true; // Mostrar detalles según el tipo de notificación switch(notification.type) { case 'assignment': // Redirigir a la orden asignada const orderId = parseInt(notification.message.match(/#(\d+)/)[1]); viewOrderDetails(orderId); break; default: alert(notification.message); showNotifications(); // Recargar notificaciones } } function clearAllNotifications() { if (confirm('¿Eliminar todas las notificaciones?')) { notifications = notifications.filter(n => !n.read); showNotifications(); } }
