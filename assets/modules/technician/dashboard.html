Mis órdenes asignadas
${createOrdersTable(technicianData.assignedOrders)}
`; } function createStatsCard(title, value, type, subtitle) { return `
${title}

${subtitle}

`; } function createOrdersTable(orders) { return `
${orders.map(order => createOrderRow(order)).join('')}

`; } function createOrderRow(order) { return ` ${order.id} ${order.client} ${order.device} ${order.problem} ${formatDate(order.assignedDate)} ${order.status} ${order.status !== 'Completada' ? ` ` : ''} `; }
